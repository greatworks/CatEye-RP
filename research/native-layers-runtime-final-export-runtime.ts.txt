    if (clampToPage !== undefined && typeof clampToPage !== 'boolean') throw new Error('Viewport clampToPage must be a boolean');

  const renderedContexts = new Map<string, RowContext>();
  const fittedPanelSizes = new Map<string, { width: number; height: number }>();

  const easingAt = (name: string, p: number): number => {
    if (name === 'linear') return p;
    if (name === 'swing') return (1 - Math.cos(Math.PI * p)) / 2;
    const family = name.replace(/^ease(?:InOut|In|Out)/, '');
    const incoming = (t: number): number => {
      const power = ({ Quad: 2, Cubic: 3, Quart: 4, Quint: 5, Expo: 6 } as Record<string, number>)[family];
      if (power) return t ** power;
      if (family === 'Sine') return 1 - Math.cos(Math.PI * t / 2);
      if (family === 'Circ') return 1 - Math.sqrt(1 - t * t);
      if (family === 'Back') return t * t * (3 * t - 2);
      if (family === 'Elastic') return t === 0 || t === 1 ? t : -(2 ** (8 * (t - 1))) * Math.sin((80 * (t - 1) - 7.5) * Math.PI / 15);
      if (family === 'Bounce') {
        // Independent piecewise polynomial matching the shipped UI easing.
        const level = t >= 7 / 11 ? 3 : t >= 3 / 11 ? 2 : t >= 1 / 11 ? 1 : 0;
        return 4 ** (level - 3) - 7.5625 * (((3 * 2 ** level - 2) / 22) - t) ** 2;
      }
      throw new Error(`Unsupported Rotate easing: ${name}`);
    };
    if (name.startsWith('easeInOut')) return p < .5 ? incoming(p * 2) / 2 : 1 - incoming(2 - p * 2) / 2;
    return name.startsWith('easeOut') ? 1 - incoming(1 - p) : incoming(p);

        if (alias === 'Window' && dot >= 0) return field === 'width' ? request.frame.window?.width : field === 'height' ? request.frame.window?.height : undefined;
        if (alias === 'Item' && dot >= 0) return itemValue(field, request.context, request.row);
        if (alias === 'TargetItem' && dot >= 0) return itemValue(field, request.scope);
        return dragValue(name) ?? read(name, request.context);
      };

      try { return expressionEvaluator(source.trim(), provider, 'motion'); }
      catch (error) { if (error instanceof Error && error.name === 'MotionEvaluationError') return _all; throw error; }
    });

    if (!Number.isFinite(number)) throw new Error('Motion expressions must produce finite numbers');

    const value = transformValue(expression, request), number = value === '' ? previous : Number(value);

  const transformBoundary = (boundary: MotionBoundary, request: TransformRequest, delta: { x: number; y: number }) => {
    const test = (node: MotionBoundary): boolean => {
      if (node.kind !== 'compare') return node.kind === 'and' ? test(node.left) && test(node.right) : test(node.left) || test(node.right);
      const left = transformNumber(node.left, request).number, right = transformNumber(node.right, request).number;
      return node.operator === '>' ? left > right : node.operator === '<' ? left < right : node.operator === '>=' ? left >= right : left <= right;
    };
    if (test(boundary)) return delta;
    const rect = widgetGeometry(request.original, request.scope)!;
    // The native compiler folds ordered leaves, including OR, only after the
    // separately evaluated predicate failed. Its first direction expression is
    // the seed for that axis. Conflicting limits must not be reordered.
    const projected: { x?: number; y?: number } = {};
    const collect = (node: MotionBoundary) => {
      if (node.kind !== 'compare') { collect(node.left); collect(node.right); return; }
      const right = transformNumber(node.right, request).number;
      const horizontal = node.direction === 'left' || node.direction === 'right', size = node.direction === 'right' ? rect.width : node.direction === 'bottom' ? rect.height : 0;
      const lower = node.operator === '>' || node.operator === '>=';
      const value = right - size + (node.operator === '>' ? 1 : node.operator === '<' ? -1 : 0);
      const key = horizontal ? 'x' : 'y', previous = projected[key] ?? transformNumber(node.left, request).number - size;
      projected[key] = lower ? Math.max(previous, value) : Math.min(previous, value);
    };
    collect(boundary);
    return { x: projected.x === undefined ? delta.x : projected.x - rect.x, y: projected.y === undefined ? delta.y : projected.y - rect.y };
  };
  const sliceRatio = (slice: TransformSlice, remaining: boolean) => slice.length ? (slice.end - slice.start) / (remaining ? slice.length - slice.start : slice.length) : 1;
  const transformMove = (slice: TransformSlice): { x: number; y: number } => {
    const request = transformRequests.get(slice.request)!, spec = request.move!, rect = widgetGeometry(request.original, request.scope)!;
    if (spec.mode !== 'delta' && spec.mode !== 'location') {
      const drag = request.frame.drag;
      let delta: { x: number; y: number };
      if (spec.mode === 'drag' || spec.mode === 'dragX' || spec.mode === 'dragY') delta = { x: spec.mode === 'dragY' ? 0 : drag?.deltaX ?? NaN, y: spec.mode === 'dragX' ? 0 : drag?.deltaY ?? NaN };
      else if (spec.mode === 'locationBeforeDrag') {
        const saved = drag?.movedWidgets[instanceKey(request.original.id, request.scope)];
        if (!saved) { if (request.frame.sourceId) releaseTransformChannel(instanceKey(request.frame.sourceId, request.context), 'move'); return { x: 0, y: 0 }; }
        delta = { x: saved.x - rect.x, y: saved.y - rect.y };
      } else if (spec.mode === 'withThis') {
        const source = request.frame.sourceId && sourceMoves.get(instanceKey(request.frame.sourceId, request.context));
        if (!source) throw new Error('Move withThis requires source movement information');
        delta = { x: source.deltaX, y: source.deltaY }; request.duration = source.duration; request.easing = source.easing;
      } else throw new Error(`Move mode ${spec.mode} requires a special layout target`);
      if (!Number.isFinite(delta.x) || !Number.isFinite(delta.y)) throw new Error('Move drag context has non-finite native deltas');
      return spec.boundary ? transformBoundary(spec.boundary, request, delta) : delta;
    }
    let delta = transformMoveResults.get(request.id);

    }

    const ratio = sliceRatio(slice, spec.mode === 'location'); delta = { x: delta.x * ratio, y: delta.y * ratio };
    if (spec.boundary) delta = transformBoundary(spec.boundary, request, delta);

  const emitTransform = (queue: Record<TransformKind, number[]>, complete: boolean) => {
    let steps = 0;
    while (Object.values(queue).some(items => items.length)) {
      if (!complete && Object.values(queue).some(items => !items.length)) break;
      if (++steps > 10000) throw new Error('Motion planner made no progress');
      const requests = (['move','resize','rotate'] as const).map(kind => queue[kind][0] && transformRequests.get(queue[kind][0])).filter((item): item is TransformRequest => !!item);
      // Retained mixed instant tails can forward the wrong request forever.
      // Reject the whole transaction instead of reproducing that nonprogress.
      if (requests.some(item => item.duration === 0) && requests.some(item => item.duration > 0)) throw new Error('Motion planner has an unsupported non-progressing instant tail');
      const duration = Math.min(...requests.map(item => item.duration - item.used)), leader = requests.find(item => item.kind === 'resize') ?? requests.find(item => item.kind === 'rotate') ?? requests[0];
      const segment: TransformSegment = { duration, easing: (requests.find(item => item.kind === 'move') ?? requests.find(item => item.kind === 'resize') ?? requests[0]).easing };
      for (const request of requests) {
        segment[request.kind] = { request: request.id, start: request.used, end: request.used + duration, length: request.duration };
        request.used += duration;
        if (request.used >= request.duration) queue[request.kind].shift();
      }
      const key = instanceKey(leader.original.id, leader.scope), job: Motion = {
        id: ++nextMotion, key, original: leader.original, action: leader.action, row: leader.row, context: leader.context, scope: leader.scope, frame: leader.frame, group: leader.group,

      };
      if (segment.rotate && !job.channels.includes(`move:${key}`)) job.channels.push(`move:${key}`);
      if (segment.resize && !job.channels.includes(`move:${key}`) && (segment.rotate || transformRequests.get(segment.resize.request)!.resize!.anchor !== 'top left')) job.channels.push(`move:${key}`);
      leader.group.pending++; motions.set(job.id, job);
      if (leader.original.layer && (segment.resize || segment.rotate)) {
        job.layerChildren = [];

          const childKey = instanceKey(child.id, leader.scope), kind = segment.resize ? 'resize' : 'rotate';
          const channels = [`${kind}:${childKey}`];

          const member: Motion = { ...job, id: ++nextMotion, key: childKey, original: child, channels, generation: motionGenerations.get(childKey) ?? 0, layerParent: job.id, layerChildren: undefined };
          leader.group.pending++; motions.set(member.id, member); job.layerChildren.push(member.id); runtime.scopes[childKey] = leader.scope;
        }
      }

    }

  const motionCanPaint = (job: Motion) => job.epoch === pageEpoch && validContext(job.scope) && job.generation === (motionGenerations.get(job.key) ?? 0);
  const shadowCenter = (w: Widget, scope: RowContext) => {

    const moved = rotationLocations.get(key);

  const arcPosition = (trajectory: MoveTargetSpec['trajectory'], x: number, y: number, progress: number) => {
    if (trajectory === 'straight' || !x || !y) return { x: x * progress, y: y * progress };
    const useX = Boolean(Number(x > 0) ^ Number(y > 0) ^ Number(trajectory === 'arcClockwise'));
    const point = (t: number) => useX ? { x: Math.sin(t * Math.PI / 2) * x, y: (1 - Math.cos(t * Math.PI / 2)) * y } : { x: (1 - Math.cos(t * Math.PI / 2)) * x, y: Math.sin(t * Math.PI / 2) * y };

    const length = (index: number) => Math.hypot(samples[index + 1].x - samples[index].x, samples[index + 1].y - samples[index].y);
    for (let n = 0; n < 10000; n++) {
      const lengths = samples.slice(1).map((_p, index) => length(index)), total = lengths.reduce((sum, value) => sum + value, 0), longest = Math.max(...lengths);
      if (longest * 100 <= total) break;
      const index = lengths.indexOf(longest), t = (samples[index].t + samples[index + 1].t) / 2; samples.splice(index + 1, 0, { t, ...point(t) });
    }
    const lengths = samples.slice(1).map((_p, index) => length(index)), total = lengths.reduce((sum, value) => sum + value, 0);
    let distance = total * progress, index = 0;
    while (index < lengths.length - 1 && distance > lengths[index]) distance -= lengths[index++];
    return point(samples[index].t + (samples[index + 1].t - samples[index].t) * distance / lengths[index]);
  };
  const paintTransformTween = (job: Motion, tween: TransformTween, progress: number) => {
    if (!motionCanPaint(job)) return;
    if (tween.kind === 'rotate') { patch(job.original, { rotation: tween.angle! + tween.delta! * progress }, job.scope); return; }
    if (tween.kind === 'orbit') {
      if (!tween.delta) {
        // Zero rotation still consumes the Layer member's resize translation,
        // but not resizeOffset or the circular-move location-cache path.
        patch(job.original, { x: tween.from.x + ((tween.move?.x ?? 0) + (tween.rotatableMove?.x ?? 0)) * progress, y: tween.from.y + ((tween.move?.y ?? 0) + (tween.rotatableMove?.y ?? 0)) * progress }, job.scope); return;
      }
      const degree = tween.delta! * progress, deltaDegree = degree - tween.previous!, w = currentWidget(job.original, job.scope), center = shadowCenter(w, job.scope);
      const next = rotatePoint(center, tween.pivot!, deltaDegree), ratio = tween.delta ? deltaDegree / tween.delta : progress - tween.previous!;
      const dx = ((tween.move?.x ?? 0) + (tween.rotatableMove?.x ?? 0) + (tween.resizeOffset?.x ?? 0)) * ratio, dy = ((tween.move?.y ?? 0) + (tween.rotatableMove?.y ?? 0) + (tween.resizeOffset?.y ?? 0)) * ratio;

      if (tween.rotatableMove) tween.rotatableMove = rotatePoint(tween.rotatableMove, { x: 0, y: 0 }, deltaDegree);

    }
    const delta = tween.kind === 'move' ? arcPosition(tween.trajectory ?? 'straight', tween.to.x - tween.from.x, tween.to.y - tween.from.y, progress) : { x: (tween.to.x - tween.from.x) * progress, y: (tween.to.y - tween.from.y) * progress };

    }
    const changes: Partial<Widget> = tween.position === false ? {} : { x: tween.from.x + delta.x, y: tween.from.y + delta.y };

    patch(job.original, changes, job.scope);
  };

    }
  };
  const tickTransform = (job: Motion, fraction: number) => {
    const progress = easingAt(job.transform!.easing === 'none' ? 'linear' : job.transform!.easing, fraction);
    // Creation order is observable. A completion can start another consumer
    // before this job's later tween writes its last frame.
    for (const tween of job.tweens ?? []) if (!tween.done) {
      paintTransformTween(job, tween, progress);
      if (fraction === 1) completeTransformTween(job, tween);
    }
    if (job.tweens?.every(tween => tween.done)) { job.status = 'done'; job.channels = []; pumpMotions(); }
  };
  const canResize = (w: Widget) => ['box','ellipse','path','text','button','placeholder','panel','image','hotspot','inlineFrame','select','input','textarea'].includes(w.type);
  const layerAngleKey = (w: Widget, scope: RowContext) => JSON.stringify(['layer', w.id, scope]);
  const layerMoveEvent = (job: Motion) => {
    geometryEvent(job.original, 'move', job.scope, job.group);
    if (!job.layerMove) return;
    const info = sourceMoves.get(job.key);
    for (const member of layerMembers(job.original, job.scope, false)) {
      if (info) sourceMoves.set(instanceKey(member.id, job.scope), { ...info });
      geometryEvent(member, 'move', job.scope, job.group);
    }
  };
  const prepareLayerMove = (job: Motion, move: { x: number; y: number }) => {
    if (!job.original.layer || !move.x && !move.y) return;
    job.layerMove = layerMembers(job.original, job.scope).map(original => {
      const key = instanceKey(original.id, job.scope), from = currentWidget(original, job.scope);
      rotationLocations.set(key, { x: (rotationLocations.get(key)?.x ?? from.x) + move.x, y: (rotationLocations.get(key)?.y ?? from.y) + move.y });
      return { original, key, from: { x: from.x, y: from.y, width: from.width, height: from.height }, generation: motionGenerations.get(key) ?? 0 };
    });
  };
  const layerPivot = (job: Motion, bounds: ItemRect, anchor: string, combined: boolean) => {
    const request = transformRequests.get(job.transform!.rotate!.request)!, spec = request.action.rotate!;
    let x = Number(transformValue({ text: spec.offsetX }, request));
    // The combined native consumer evaluates Y on the original event info,
    // unlike X on its target-bound copy. It does not invent a member Target.
    let y = Number(transformValue({ text: spec.offsetY }, combined ? { ...request, frame: { ...request.frame, targetId: request.frame.originalTargetId } } : request));
    if (anchor.includes('left')) x -= bounds.width / 2; if (anchor.includes('right')) x += bounds.width / 2;
    if (anchor.includes('top')) y -= bounds.height / 2; if (anchor.includes('bottom')) y += bounds.height / 2;
    if (![x, y].every(Number.isFinite)) throw new Error('Layer rotation requires finite expressions');
    return { x: bounds.x + bounds.width / 2 + x, y: bounds.y + bounds.height / 2 + y };
  };
  const startLayerMember = (job: Motion) => {
    const plan = job.layerPlan, segment = job.transform!;
    if (!plan) { job.status = 'done'; job.channels = []; return; }
    job.started = Date.now(); job.duration = segment.duration; job.tweens = [];
    const initial = currentWidget(job.original, job.scope);
    if (initial.layer) {
      job.status = 'done'; job.channels = [];
      geometryEvent(job.original, segment.resize ? 'resize' : 'rotate', job.scope, job.group);
      return;
    }
    pinMotionParents(initial, job.scope);
    let delta = { x: 0, y: 0 };
    if (segment.resize) {
      const rect = offsetBounds(initial, job.scope), location = offsetBounds(initial, job.scope, true), size = widgetGeometry(initial, job.scope)!;
      const axisDelta = (position: number, length: number, center: number, origin: number, extent: number, percent: number, leading: boolean, trailing: boolean) => {
        if (leading) { const value = (position - origin) * percent; return value + (!canResize(initial) && Math.round(value) ? length * percent : 0); }
        if (trailing) { if (canResize(initial)) return (position - origin - extent) * percent; const value = Math.round(position + length - origin - extent) * percent; return value ? value - length * percent : 0; }
        if (canResize(initial)) return (center - origin - extent / 2) * percent - length * percent / 2;
        const value = Math.round(center - origin - extent / 2) * percent;
        return value > 0 ? value + Math.abs(length * percent / 2) : value < 0 ? value - Math.abs(length * percent / 2) : 0;
      };
      const anchor = plan.resizeAnchor!;
      delta = {
        x: axisDelta(location.x, size.width, rect.x + rect.width / 2, plan.bounds.x, plan.bounds.width, plan.percentX, anchor.includes('left'), anchor.includes('right')),
        y: axisDelta(location.y, size.height, rect.y + rect.height / 2, plan.bounds.y, plan.bounds.height, plan.percentY, anchor.includes('top'), anchor.includes('bottom')),
      };
      if (canResize(initial)) {
        const width = size.width * (1 + plan.percentX), height = size.height * (1 + plan.percentY);
        if (!Number.isFinite(width) || !Number.isFinite(height) || width < 0 || height < 0) throw new Error('Layer resize produced invalid member dimensions');
        if (initial.type === 'panel') patch(initial, { fitToContent: false }, job.scope);
        resizeTargets.set(job.key, { width, height }); resizingRects.set(job.key, size);
        const tween: TransformTween = { kind: 'resize', done: false, from: { ...initial }, to: { ...initial, width, height }, position: false, releaseMove: true };
        job.tweens.push(tween);
        if (!job.duration) { paintTransformTween(job, tween, 1); completeTransformTween(job, tween); }
      } else releaseTransformChannel(job.key, 'resize');
    }
    if (segment.rotate) {
      if (!segment.resize) geometryEvent(job.original, 'rotate', job.scope, job.group);
      const current = currentWidget(job.original, job.scope), angle = liveAngle(current.rotation);
      const pivot = segment.resize ? layerPivot(job, plan.bounds, plan.resizeAnchor!, true) : { ...plan.pivot! };
      prepareTransformMove(job, plan.move);
      job.tweens.push({ kind: 'orbit', done: false, from: { ...current }, to: { ...current }, delta: plan.delta!, previous: 0, pivot, move: { ...plan.move }, rotatableMove: delta, ...(plan.resizeOffset ? { resizeOffset: { ...plan.resizeOffset } } : {}) });
      if (canOrient(current) && plan.delta) {
        rotationTargets.set(job.key, angle + plan.delta);
        job.tweens.push({ kind: 'rotate', done: false, from: { ...current }, to: { ...current }, angle, delta: plan.delta });
      } else releaseTransformChannel(job.key, 'rotate');
    } else {
      const move = { x: plan.move.x + delta.x, y: plan.move.y + delta.y };
      prepareTransformMove(job, move); transformLocation(job, move);
      job.tweens.push({ kind: 'move', done: false, from: { ...initial }, to: { ...initial, x: initial.x + move.x, y: initial.y + move.y }, trajectory: 'straight' });
    }
    if (!job.duration) tickTransform(job, 1);
  };
  const startLayerTransform = (job: Motion) => {
    const segment = job.transform!, members = job.layerChildren ?? [];
    if (!members.length) { job.status = 'done'; job.channels = []; return; }
    const initial = currentWidget(job.original, job.scope), bounds = offsetBounds(initial, job.scope);
    patch(job.original, bounds, job.scope);
    const plan: NonNullable<Motion['layerPlan']> = { bounds, percentX: 0, percentY: 0, move: { x: 0, y: 0 } };
    if (segment.resize) {
      const request = transformRequests.get(segment.resize.request)!, spec = request.resize!, ratio = sliceRatio(segment.resize, true);
      const width = transformSize(spec.width, request, bounds.width), height = transformSize(spec.height, request, bounds.height);
      plan.percentX = Math.round(width - bounds.width) / bounds.width * ratio; plan.percentY = Math.round(height - bounds.height) / bounds.height * ratio;
      if (![plan.percentX, plan.percentY].every(Number.isFinite)) throw new Error('Layer resize requires finite nonzero bounds');
      plan.resizeAnchor = spec.anchor;
      plan.resizeOffset = { x: (spec.anchor.includes('left') ? -1 : spec.anchor.includes('right') ? 1 : 0) * plan.percentX * bounds.width / 2, y: (spec.anchor.includes('top') ? -1 : spec.anchor.includes('bottom') ? 1 : 0) * plan.percentY * bounds.height / 2 };
      geometryEvent(job.original, 'resize', job.scope, job.group);
    }
    if (segment.move) { plan.move = transformMove(segment.move); prepareTransformMove(job, plan.move); geometryEvent(job.original, 'move', job.scope, job.group); }
    if (segment.rotate) {
      const request = transformRequests.get(segment.rotate.request)!, spec = request.action.rotate!, angleKey = layerAngleKey(job.original, job.scope);
      const degree = parseFloat(transformValue({ text: spec.degrees }, request)) * (segment.resize || spec.clockwise ? 1 : -1);
      const previous = groupAngles.get(angleKey) ?? initial.rotation, next = spec.mode === 'to' ? degree : previous + degree;
      plan.delta = (next - previous) * sliceRatio(segment.rotate, true); groupAngles.set(angleKey, next);
      if (![degree, plan.delta].every(Number.isFinite)) throw new Error('Layer rotation requires finite expressions');
      if (!segment.resize) plan.pivot = layerPivot(job, bounds, spec.anchor, false);
      geometryEvent(job.original, 'rotate', job.scope, job.group);
      if (segment.resize) geometryEvent(job.original, 'rotate', job.scope, job.group);
    }
    for (const id of members) { const member = motions.get(id); if (member) member.layerPlan = structuredClone(plan); }
    job.status = 'done'; job.channels = [];
  };
  const startTransform = (job: Motion) => {
    const segment = job.transform!;
    if (job.layerParent !== undefined) { startLayerMember(job); return; }
    if (job.original.layer && (segment.resize || segment.rotate)) { startLayerTransform(job); return; }
    if (!segment.move && !segment.resize && segment.rotate) {
      const request = transformRequests.get(segment.rotate.request)!, original = request.action.rotate!;
      const requested = parseFloat(transformValue({ text: original.degrees }, request)) * (original.clockwise ? 1 : -1), angle = liveAngle(currentWidget(job.original, job.scope).rotation);
      const delta = (original.mode === 'to' ? requested - angle : requested) * sliceRatio(segment.rotate, original.mode === 'to');
      job.action = { ...request.action, rotate: { ...original, mode: 'by', degrees: String(delta), clockwise: true, easing: segment.easing as typeof original.easing, duration: segment.duration } };
      job.transform = undefined; startMotion(job); return;
    }
    job.tweens = []; job.started = Date.now(); job.duration = segment.duration;
    const initial = currentWidget(job.original, job.scope), publicRect = widgetGeometry(job.original, job.scope)!;
    pinMotionParents(initial, job.scope);
    const rotationRequest = segment.rotate && transformRequests.get(segment.rotate.request)!;
    const earlyDegree = rotationRequest && !segment.resize ? parseFloat(transformValue({ text: rotationRequest.action.rotate!.degrees }, rotationRequest)) : undefined;
    let move = segment.move ? transformMove(segment.move) : { x: 0, y: 0 };
    if (segment.move) {
      const request = transformRequests.get(segment.move.request)!;
      if (request.move!.mode === 'withThis') { segment.easing = request.easing; job.duration = request.duration; }
      prepareTransformMove(job, move);
      prepareLayerMove(job, move);
    }
    let resized: ItemRect | undefined, firstDelta = { x: 0, y: 0 }, anchorDelta = { x: 0, y: 0 };
    const resizeRequest = segment.resize && transformRequests.get(segment.resize.request)!;
    if (resizeRequest) {
      if (!['box','ellipse','path','text','button','placeholder','panel','image','hotspot','inlineFrame','select','input','textarea'].includes(initial.type)) {
        releaseTransformChannel(job.key, 'resize'); segment.resize = undefined;
      } else {
        const spec = resizeRequest.resize!, ratio = sliceRatio(segment.resize!, true);
        const requestedW = transformSize(spec.width, resizeRequest, publicRect.width), requestedH = transformSize(spec.height, resizeRequest, publicRect.height);
        firstDelta = { x: Math.round(requestedW - publicRect.width) * ratio, y: Math.round(requestedH - publicRect.height) * ratio };
        const dw = firstDelta.x * ratio, dh = firstDelta.y * ratio;
        anchorDelta = { x: spec.anchor.includes('left') ? 0 : spec.anchor.includes('right') ? -dw : -dw / 2, y: spec.anchor.includes('top') ? 0 : spec.anchor.includes('bottom') ? -dh : -dh / 2 };
        resized = { x: initial.x + move.x + anchorDelta.x, y: initial.y + move.y + anchorDelta.y, width: publicRect.width + dw, height: publicRect.height + dh };
      }
    }
    let orbitHandlesMove = false;
    if (segment.rotate) {
      const request = transformRequests.get(segment.rotate.request)!, spec = request.action.rotate!;
      let offsetX = Number(transformValue({ text: spec.offsetX }, request)), offsetY = Number(transformValue({ text: spec.offsetY }, request));
      const bounds = segment.resize ? offsetBounds(initial, job.scope) : initial;
      if (spec.anchor.includes('left')) offsetX -= bounds.width / 2; if (spec.anchor.includes('right')) offsetX += bounds.width / 2;
      if (spec.anchor.includes('top')) offsetY -= bounds.height / 2; if (spec.anchor.includes('bottom')) offsetY += bounds.height / 2;
      geometryEvent(job.original, 'rotate', job.scope, job.group);
      const current = currentWidget(job.original, job.scope), angle = liveAngle(current.rotation);
      const requested = (earlyDegree ?? parseFloat(transformValue({ text: spec.degrees }, request))) * (segment.resize || spec.clockwise ? 1 : -1);
      const delta = (spec.mode === 'to' ? requested - angle : requested) * sliceRatio(segment.rotate, spec.mode === 'to');
      if (![delta, offsetX, offsetY].every(Number.isFinite)) throw new Error('Motion rotation must be finite');
      if (offsetX || offsetY) {
        orbitHandlesMove = true;
        if (resized && (!spec.anchor.includes('left') && !spec.anchor.includes('right') || !spec.anchor.includes('top') && !spec.anchor.includes('bottom'))) throw new Error('This Resize/Rotate anchor has non-finite native orbit geometry');
        const offset = rotatePoint({ x: offsetX, y: offsetY }, { x: 0, y: 0 }, angle), center = shadowCenter(current, job.scope);
        move = { x: move.x + anchorDelta.x, y: move.y + anchorDelta.y };
        if (delta || move.x || move.y) job.tweens.push({ kind: 'orbit', done: false, from: { ...initial }, to: { ...initial }, angle, delta, previous: 0, pivot: { x: center.x + offset.x, y: center.y + offset.y }, move,
          ...(resized ? { resizeOffset: { x: (spec.anchor.includes('left') ? -1 : 1) * firstDelta.x / 2, y: (spec.anchor.includes('top') ? -1 : 1) * firstDelta.y / 2 } } : {}) });
        else releaseTransformChannel(job.key, 'move');
        if (resized) { resized.x = initial.x; resized.y = initial.y; }
      } else if (!segment.move || resized) releaseTransformChannel(job.key, 'move');
      if (delta && (!offsetX && !offsetY || canOrient(current))) {
        rotationTargets.set(job.key, angle + delta);
        job.tweens.push({ kind: 'rotate', done: false, from: { ...initial }, to: { ...initial }, angle, delta });
      } else releaseTransformChannel(job.key, 'rotate');
    }
    if (resized && !firstDelta.x && !firstDelta.y) {
      // No resize CSS is created. A merged Move still animates independently,
      // but the Resize channel and its event complete immediately.
      if (!segment.rotate && (segment.move || resizeRequest!.resize!.anchor !== 'top left')) {
        transformLocation(job, move);
        job.tweens.push({ kind: 'move', done: false, from: { ...initial }, to: { ...initial, x: initial.x + move.x, y: initial.y + move.y }, trajectory: segment.move ? transformRequests.get(segment.move.request)!.move!.trajectory : 'straight' });
        if (!job.duration) tickTransform(job, 1);
      }
      if (segment.move) geometryEvent(job.original, 'move', job.scope, job.group);
      releaseTransformChannel(job.key, 'resize'); geometryEvent(job.original, 'resize', job.scope, job.group);
    } else if (resized) {
      if (resized.width || resized.height) { resizeTargets.set(job.key, { width: resized.width, height: resized.height }); transformZeroSizes.delete(job.key); }
      else transformZeroSizes.set(job.key, { width: publicRect.width, height: publicRect.height });
      const handlesMove = !orbitHandlesMove;
      const moves = Boolean(segment.move || segment.rotate || resizeRequest!.resize!.anchor !== 'top left');
      if (handlesMove && (resized.x || resized.y)) transformLocation(job, { x: resized.x - initial.x, y: resized.y - initial.y });
      resizingRects.set(job.key, publicRect);
      if (segment.move) geometryEvent(job.original, 'move', job.scope, job.group);
      if (resized.width !== publicRect.width || resized.height !== publicRect.height || resized.x !== initial.x || resized.y !== initial.y) job.tweens.push({ kind: 'resize', done: false, from: { x: initial.x, y: initial.y, width: initial.width, height: initial.height }, to: resized, position: handlesMove, releaseMove: moves });
      else { releaseTransformChannel(job.key, 'resize'); geometryEvent(job.original, 'resize', job.scope, job.group); resizingRects.delete(job.key); }
    } else if (segment.move && !orbitHandlesMove) {
      const request = transformRequests.get(segment.move.request)!;
      if (!move.x && !move.y) {
        if (!segment.rotate) sourceMoves.set(job.key, { deltaX: 0, deltaY: 0, easing: 'none', duration: 0 });
        geometryEvent(job.original, 'move', job.scope, job.group); releaseTransformChannel(job.key, 'move');
      } else {
        transformLocation(job, move);
        job.tweens.push({ kind: 'move', done: false, from: { ...initial }, to: { ...initial, x: initial.x + move.x, y: initial.y + move.y }, trajectory: request.move!.trajectory });
        if (!job.duration) tickTransform(job, 1);
        layerMoveEvent(job);
      }
    }
    if (!job.tweens.length) { job.status = 'done'; job.channels = []; pumpMotions(); }
    else if (!job.duration && job.status !== 'done') tickTransform(job, 1);
  };
  const paintMotion = (job: Motion, fraction: number, moveOnly = false) => {
    const degree = job.delta! * fraction;
    const members = job.members ?? [{ original: job.original, key: job.key, generation: job.generation, angle: job.angle!, orient: job.orient! }];
    for (const member of members) if (job.epoch === pageEpoch && validContext(job.scope) && member.generation === (motionGenerations.get(member.key) ?? 0)) {
      const w = currentWidget(member.original, job.scope), changes: Partial<Widget> = {};
      if (job.pivot) {
        const center = shadowCenter(w, job.scope), next = rotatePoint(center, job.pivot, degree - job.previous!);
        changes.x = w.x + next.x - center.x; changes.y = w.y + next.y - center.y;
        const previousLocation = rotationLocations.get(member.key), shadow = w.style.outerShadow, override = runtime.widgetOverrides[member.key];
        const initialX = previousLocation?.x ?? w.x + (!override || override.x === undefined && override.y === undefined ? shadow?.on && shadow.offsetX < 0 ? shadow.offsetX - shadow.blurRadius : 0 : 0);
        const initialY = previousLocation?.y ?? w.y + (!override || override.x === undefined && override.y === undefined ? shadow?.on && shadow.offsetY < 0 ? shadow.offsetY - shadow.blurRadius : 0 : 0);
        rotationLocations.set(member.key, { x: initialX + next.x - center.x, y: initialY + next.y - center.y });
      }
      if (member.orient && !moveOnly) changes.rotation = member.angle + degree;
      patch(w, changes, job.scope);
    }
    job.previous = degree;
  };
  const finishMotion = (job: Motion) => {
    const fit = motionCanPaint(job) && job.delta !== 0;
    const before = fit ? ancestorBounds(job.original, job.scope) : undefined;
    job.status = 'done'; job.channels = [];
    // Native completion releases the next queued rotation before fitting parents.
    pumpMotions();
    if (fit) for (const [index, member] of (job.members ?? [{ original: job.original }]).entries()) fitFrom(member.original, job.scope, job.group, index ? ancestorBounds(member.original, job.scope) : before!, true, true);
  };
  const completeMotion = (job: Motion) => {
    if (job.pivot && job.orient && !job.groupId) {
      const before = ancestorBounds(job.original, job.scope);
      paintMotion(job, 1, true);
      job.channels = job.channels.filter(channel => !channel.startsWith('move:'));
      pumpMotions();
      if (motionCanPaint(job)) {
        fitFrom(job.original, job.scope, job.group, before, true, true);
        // circularMove completes before the separately created orientation
        // tween. An instant queued Move observes the previous orientation.
        if (!job.duration) rotationTargets.set(job.key, job.angle! + job.delta!);
        patch(job.original, { rotation: job.angle! + job.delta! }, job.scope);
      }
    } else paintMotion(job, 1);
    finishMotion(job);
  };
  const startMotion = (job: Motion) => {
    job.status = 'active';
    // A queued native action resolves a replacement element when it starts;
    // only an already-running tween remains attached to its old generation.
    job.generation = motionGenerations.get(job.key) ?? 0;
    if (!motionCanPaint(job)) { finishMotion(job); return; }
    if (job.transform) { startTransform(job); return; }
    if (job.action.type !== 'rotate') {
      action(job.action, job.row, job.context, job.frame, job.group, job.scope);
      if (job.action.type === 'move' && rotationLocations.has(job.key)) { const w = baseWidget(job.original, job.scope); rotationLocations.set(job.key, { x: w.x, y: w.y }); }
      job.status = 'done'; job.channels = []; pumpMotions(); return;
    }
    const spec = job.action.rotate; if (!spec) throw new Error('Missing Rotate action specification');
    const requested = parseFloat(resolve(spec.degrees, job.row, job.context, job.scope, job.frame)) * (spec.clockwise ? 1 : -1);
    let offsetX = Number(resolve(spec.offsetX, job.row, job.context, job.scope, job.frame));
    let offsetY = Number(resolve(spec.offsetY, job.row, job.context, job.scope, job.frame));
    if (![requested, offsetX, offsetY, spec.duration].every(Number.isFinite) || spec.duration < 0) throw new Error('Rotate expressions and duration must be finite');
    if (job.groupId) {
      const members = flatGroups.get(job.groupId)!.filter(w => !isUnplaced(w, job.scope));
      if (!members.length) { job.status = 'done'; return; }
      const boxes = members.map(w => offsetBounds(currentWidget(w, job.scope), job.scope));
      const left = Math.min(...boxes.map(b => b.x)), top = Math.min(...boxes.map(b => b.y));
      const width = Math.max(...boxes.map(b => b.x + b.width)) - left, height = Math.max(...boxes.map(b => b.y + b.height)) - top;
      if (spec.anchor.includes('left')) offsetX -= width / 2; if (spec.anchor.includes('right')) offsetX += width / 2;
      if (spec.anchor.includes('top')) offsetY -= height / 2; if (spec.anchor.includes('bottom')) offsetY += height / 2;
      const oldAngle = groupAngles.get(job.key) ?? 0;
      job.delta = spec.mode === 'to' ? requested - oldAngle : requested; job.angle = oldAngle; job.previous = 0;
      groupAngles.set(job.key, oldAngle + job.delta);
      job.pivot = { x: left + width / 2 + offsetX, y: top + height / 2 + offsetY };
      job.members = [];
      if (job.delta) for (const original of members) pinMotionParents(original, job.scope);
      for (const original of members) {
        geometryEvent(original, 'rotate', job.scope, job.group);
        if (job.epoch !== pageEpoch || job.status !== 'active') { job.status = 'done'; return; }
        const w = currentWidget(original, job.scope), key = instanceKey(w.id, job.scope), angle = liveAngle(w.rotation), orient = canOrient(w);
        job.members.push({ original, key, angle, orient, generation: motionGenerations.get(key) ?? 0 });
        if (orient && job.delta) rotationTargets.set(key, angle + job.delta);
      }
      job.started = Date.now(); job.duration = spec.easing === 'none' ? 0 : spec.duration;
      if (!job.delta || !job.duration) { if (job.delta) paintMotion(job, 1); finishMotion(job); }
      return;
    }
    const initial = currentWidget(job.original, job.scope);
    if (spec.anchor.includes('left')) offsetX -= initial.width / 2;
    if (spec.anchor.includes('right')) offsetX += initial.width / 2;
    if (spec.anchor.includes('top')) offsetY -= initial.height / 2;
    if (spec.anchor.includes('bottom')) offsetY += initial.height / 2;
    geometryEvent(job.original, 'rotate', job.scope, job.group);
    if (!motionCanPaint(job) || job.status !== 'active') { job.status = 'done'; return; }
    const w = currentWidget(job.original, job.scope), angle = liveAngle(w.rotation);
    job.angle = angle; job.delta = spec.mode === 'to' ? requested - angle : requested; job.previous = 0;
    job.orient = !offsetX && !offsetY || canOrient(w);
    if (job.delta) pinMotionParents(w, job.scope);
    if (offsetX || offsetY) {
      const offset = rotatePoint({ x: offsetX, y: offsetY }, { x: 0, y: 0 }, angle), center = shadowCenter(w, job.scope);
      job.pivot = { x: center.x + offset.x, y: center.y + offset.y };
      if (!job.orient) job.channels = [`move:${job.key}`];
    } else job.channels = [`rotate:${job.key}`];
    if (!job.delta) { finishMotion(job); return; }
    job.started = Date.now(); job.duration = spec.easing === 'none' ? 0 : spec.duration;
    if (job.orient && (!job.pivot || job.duration)) rotationTargets.set(job.key, angle + job.delta);
    if (!job.duration) completeMotion(job);
  };
  const pumpMotions = (): void => {
    if (++motionDepth > 64) { motionDepth--; throw new Error('Widget geometry event cycle limit exceeded'); }
    try {
      const blocked = new Set<string>();
      for (const job of motions.values()) {
        if (job.status === 'done') continue;
        if (job.layerParent !== undefined && motions.get(job.layerParent)?.status !== 'done' && !job.layerPlan) {
          for (const channel of job.channels) blocked.add(channel);
          continue;
        }
        if (job.status === 'queued' && !job.channels.some(channel => blocked.has(channel))) {
          if (job.group === runningTaskGroup) startMotion(job);
          else {
            // Completion can release work submitted by a different trigger.
            // Its expression/event failure belongs to that trigger, never the
            // successfully completed tween which happened to release its queue.
            try { atomic(() => startMotion(job)); }
            catch (error) {
              motionFailures.set(job.group, error);
              for (const pending of motions.values()) if (pending.group === job.group) { pending.status = 'done'; pending.channels = []; }
            }
          }
        }
        if ((job.status as Motion['status']) !== 'done') for (const channel of job.channels) blocked.add(channel);
      }
    } finally { motionDepth--; }
  };
  const addMotion = (a: Action, original: Widget, row: Row | undefined, context: RowContext, scope: RowContext, frame: EventFrame, group: TaskGroup, groupId?: string) => {
    if (original.layer && (a.type === 'move' || a.type === 'resize')) {
      const w = currentWidget(original, scope);
      const spec: MoveTargetSpec | ResizeTargetSpec = a.type === 'move'
        ? { target: { kind: 'widget', id: original.id }, mode: 'location', x: { text: String(a.x ?? w.x) }, y: { text: String(a.y ?? w.y) }, trajectory: 'straight', easing: 'none', duration: 0 }
        : { target: { kind: 'widget', id: original.id }, width: { text: String(a.x ?? w.width) }, height: { text: String(a.y ?? w.height) }, anchor: 'top left', easing: 'none', duration: 0 };
      addTransform(a, original, row, context, scope, frame, group, spec); return;
    }
    if (motions.size >= 10000) throw new Error('Pending motion limit of 10000 exceeded');
    const key = groupId ? JSON.stringify(['flat-group', groupId, scope]) : instanceKey(original.id, scope), timed = a.type === 'rotate' && a.rotate?.easing !== 'none' && (a.rotate?.duration ?? 0) > 0;
    for (const member of groupId ? flatGroups.get(groupId)! : [original]) runtime.scopes[instanceKey(member.id, scope)] = scope;
    const job: Motion = { id: ++nextMotion, key, original, action: structuredClone(a), row, context, scope, frame, group,
      generation: motionGenerations.get(key) ?? 0, epoch: pageEpoch, status: timed ? 'batched' : 'queued',
      channels: a.type === 'rotate' ? [`rotate:${key}`, `move:${key}`] : [`${a.type}:${key}`], groupId };
    if (groupId) for (const member of flatGroups.get(groupId)!) { const memberKey = instanceKey(member.id, scope); job.channels.push(`rotate:${memberKey}`, `move:${memberKey}`); }
    group.pending++; motions.set(job.id, job);
    // A timed Rotate is flushed after the synchronous event chain. Instant Move
    // keeps its existing behavior and is not retroactively merged into that batch.
    if (!timed) {
      if (a.type !== 'rotate' && ![...motions.values()].some(other => other !== job && (other.status === 'active' || other.status === 'queued' && other.layerParent !== undefined) && other.channels.some(c => job.channels.includes(c)))) startMotion(job);
      else pumpMotions();
    }
  };
  const flushMotions = () => {
    const batches = [...transformBatches.values()]; transformBatches.clear();
    for (const batch of batches) for (const queues of batch.values()) emitTransform(queues, true);
    for (const job of motions.values()) if (job.status === 'batched') job.status = 'queued'; pumpMotions();
  };
  const flushTransformBatch = (frame: EventFrame) => {
    const batch = frame.motionBatch === undefined ? undefined : transformBatches.get(frame.motionBatch);
    if (!batch) return; transformBatches.delete(frame.motionBatch!);
    for (const queues of batch.values()) emitTransform(queues, true);
  };
  const commitMotions = () => {
    for (const [group, error] of motionFailures) if (!group.settled) { group.settled = true; group.reject(error); }
    motionFailures.clear();
    for (const [id, job] of motions) if (job.status === 'done') { motions.delete(id); finishTask(job.group); }
    const retained = new Set<number>();
    for (const batch of transformBatches.values()) for (const queues of batch.values()) for (const ids of Object.values(queues)) for (const id of ids) retained.add(id);
    for (const job of motions.values()) for (const kind of ['move','resize','rotate'] as const) if (job.transform?.[kind]) retained.add(job.transform[kind]!.request);
    for (const id of transformRequests.keys()) if (!retained.has(id)) { transformRequests.delete(id); transformMoveResults.delete(id); }
    const active = [...motions.values()].filter(job => job.status === 'active' && job.duration);
    if (disposed || !active.length) { if (motionTimer !== undefined) clearTimeout(motionTimer); motionTimer = undefined; return; }
    if (motionTimer !== undefined) return;
    const delay = Math.max(0, Math.min(16, ...active.map(job => job.started! + job.duration! - Date.now())));
    motionTimer = setTimeout(() => {
      motionTimer = undefined;
      for (const item of active) queueTask(item.group, () => {
        const job = motions.get(item.id); if (!job || job.status !== 'active') return;
        const fraction = Math.min(1, (Date.now() - job.started!) / job.duration!);
        if (job.transform) tickTransform(job, fraction);
        else if (fraction === 1) completeMotion(job);
        else paintMotion(job, easingAt(job.action.rotate!.easing, fraction));
        notify(); settle(job.group);
      });
    }, delay);
  };

  const directTransformMove = (a: Action, original: Widget, delta: { x: number; y: number }, spec: MoveTargetSpec, row: Row | undefined, context: RowContext, frame: EventFrame, group: TaskGroup) => {
    if (motions.size >= 10000) throw new Error('Pending motion limit of 10000 exceeded');
    const scope = widgetContext(original.id, context), key = instanceKey(original.id, scope), initial = currentWidget(original, scope);
    if (!delta.x && !delta.y) {
      sourceMoves.set(key, { deltaX: 0, deltaY: 0, easing: 'none', duration: 0 });
      geometryEvent(original, 'move', scope, group); releaseTransformChannel(key, 'move'); return;
    }
    const duration = spec.easing === 'none' || !delta.x && !delta.y ? 0 : spec.duration || 500;
    // compressMove invokes public moveBy directly. It neither joins the event
    // planner nor waits for an existing channel, but completion still releases
    // that instance's Move queue. The missing trajectory takes the native arc.
    const job: Motion = { id: ++nextMotion, key, original, action: a, row, context, scope, frame, group, generation: motionGenerations.get(key) ?? 0, epoch: pageEpoch, status: 'active', channels: [], started: Date.now(), duration,
      transform: { easing: spec.easing, duration }, tweens: [{ kind: 'move', done: false, from: { ...initial }, to: { ...initial, x: initial.x + delta.x, y: initial.y + delta.y }, trajectory: 'arcAntiClockwise' }] };
    group.pending++; motions.set(job.id, job); runtime.scopes[key] = scope; pinMotionParents(initial, scope);
    prepareTransformMove(job, delta); prepareLayerMove(job, delta); transformLocation(job, delta);
    if (!duration) tickTransform(job, 1);
    layerMoveEvent(job);
  };
  const compressTransform = (a: Action, spec: MoveTargetSpec, row: Row | undefined, context: RowContext, frame: EventFrame, group: TaskGroup) => {
    const source = frame.sourceId ? widgets.get(frame.sourceId) : undefined;
    if (!source) throw new Error('Move widgetsBelow/widgetsRight requires a widget event source');
    let parent = containers.get(source.id), container = parent && widgets.get(parent.id);
    const layerAncestors = new Set<string>();
    for (let ancestor = parent; ancestor; ancestor = containers.get(ancestor.id)) if (widgets.get(ancestor.id)?.layer) layerAncestors.add(ancestor.id);
    while (container?.layer && !baseWidget(container, context).hidden && !isUnplaced(container, context, true)) {
      parent = containers.get(container.id); container = parent && widgets.get(parent.id);
    }
    // Table/menu/tree descendants are not native state/item containers. Their
    // generated wrapper geometry has no equivalent in the saved widget graph.
    if (container && !container.layer && container.type !== 'panel' && container.type !== 'repeater') throw new Error('Relative widget compression requires a page, Layer, panel state, or repeater item container');
    const siblings = container?.layer?.widgets ?? (container?.type === 'panel' ? container.states?.find(state => state.id === parent!.stateId)?.widgets ?? [] : container?.repeater ? container.repeater.template : doc.pages.find(page => page.id === runtime.pageId)!.widgets);
    const w = currentWidget(source, context), key = instanceKey(source.id, context), current = offsetBounds(w, context);
    const old = frame.event === 'resize' ? resizingRects.get(key) ?? offsetBounds(w, context, true) : offsetBounds(w, context, true);
    const vertical = spec.target.kind === 'widgetsBelow', sizeAxis = vertical ? 'height' : 'width', crossSize = vertical ? 'width' : 'height';
    const threshold = (vertical ? old.y : old.x) + old[sizeAxis], clampStart = vertical ? w.x : w.y, clampEnd = clampStart + Math.max(old[crossSize], current[crossSize]);
    const request: TransformRequest = { id: 0, kind: 'move', original: source, action: a, row, context, scope: context, frame, group, used: 0, duration: 0, easing: spec.easing };
    const x = spec.mode === 'delta' ? Number(transformValue(spec.x, request)) : NaN, y = spec.mode === 'delta' ? Number(transformValue(spec.y, request)) : NaN;
    const magnitude = frame.event === 'resize' ? current[sizeAxis] - old[sizeAxis] : 1, explicit = vertical ? y : x;
    const primary = Number.isNaN(explicit) ? magnitude : explicit * Math.sign(magnitude), complementary = Number.isNaN(vertical ? x : y) ? 0 : vertical ? x : y;
    if (![primary, complementary, threshold, clampStart, clampEnd].every(Number.isFinite)) throw new Error('Relative widget compression requires finite geometry');
    // Current state DOMs in this renderer have their owning panel's dimensions;
    // no independent native state-size or fixed-position metadata is invented.
    const compress = (items: Widget[], nested: boolean): boolean => {
      const selected: Widget[] = []; let all = true;
      for (const candidate of items) {
        if (candidate.id === source.id) { all = false; continue; }
        const child = currentWidget(candidate, context), location = offsetBounds(child, context, true);
        const leading = vertical ? location.y : location.x, cross = vertical ? location.x : location.y;
        const matches = leading >= threshold && cross + child[crossSize] > clampStart && cross < clampEnd;
        if (child.layer) {
          const bounds = offsetBounds(child, context), end = (vertical ? bounds.y : bounds.x) + bounds[sizeAxis], crossStart = vertical ? bounds.x : bounds.y;
          if (end < threshold || crossStart + bounds[crossSize] <= clampStart || crossStart >= clampEnd) { all = false; continue; }
          if (matches && !layerAncestors.has(child.id) || compress(child.layer.widgets, true)) selected.push(candidate);
          else all = false;
        } else if (matches) selected.push(candidate);
        else all = false;
      }
      if (nested && all) return true;
      for (const candidate of selected) directTransformMove(a, candidate, vertical ? { x: complementary, y: primary } : { x: primary, y: complementary }, spec, row, context, frame, group);
      return false;
    };
    compress(siblings, false);
    // The native helper pushes a non-fit item immediately, even when the
    // selected siblings are animated. A containing panel stops this step.
    const item = context.at(-1);
    if (container?.repeater && item?.repeaterId === container.id && !container.repeater.fitToContent) {
      const scope = context.slice(0, -1), size = runtime.itemLayouts[instanceKey(container.id, scope)]?.[item.rowId];
      if (size) setItemSize(container, scope, item.rowId, Math.max(0, size.width + (vertical ? 0 : primary)), Math.max(0, size.height + (vertical ? primary : 0)), group);
    }
  };

  const action = (a: Action, row: Row | undefined, context: RowContext, frame: EventFrame, group: TaskGroup, motionScope?: RowContext): void => {
    frame = { ...frame, originalTargetId: frame.targetId, targetId: a.targetId && (widgets.has(a.targetId) || flatGroups.has(a.targetId)) ? a.targetId : undefined };
    if (a.type === 'move' && a.move || a.type === 'resize' && a.resize) {
      const specs = a.type === 'move' ? a.move!.targets : a.resize!.targets;
      for (const spec of specs) {
        if (a.type === 'move' && (spec.target.kind === 'widgetsBelow' || spec.target.kind === 'widgetsRight')) {
          compressTransform(a, spec as MoveTargetSpec, row, context, frame, group); continue;
        }
        if (a.type === 'resize' && spec.target.kind === 'thisItem') {
          const item = context.at(-1), repeater = item && widgets.get(item.repeaterId), scope = context.slice(0, -1);
          const old = repeater && item ? runtime.itemLayouts[instanceKey(repeater.id, scope)]?.[item.rowId] : undefined;
          // The retained thisItem branch omits Target while evaluating size and
          // does not consume anchor, easing, duration, or animation channels.
          // Page events still evaluate the dimensions, then no-op without an
          // item. Neither This nor Target needs a manufactured widget.
          const request: TransformExpressionContext = { row, context, scope: context, frame: { ...frame, targetId: undefined } };
          const size = spec as ResizeTargetSpec;
          const width = transformSize(size.width, request, old?.width ?? 0), height = transformSize(size.height, request, old?.height ?? 0);
          if (old && repeater && item) setItemSize(repeater, scope, item.rowId, width, height, group); continue;
        }
        const id = spec.target.kind === 'widget' ? spec.target.id : spec.target.kind === 'this' ? frame.sourceId : undefined;
        if (!id) throw new Error(`Motion target ${spec.target.kind} requires a connected motion event context`);
        const original = widgets.get(id); if (!original) { if (flatGroups.has(id)) throw new Error('Move/Resize of a flat group requires native Layer geometry'); continue; }
        if (pageOwners.get(id) !== runtime.pageId) continue;
        for (const scope of targetContexts(id, context)) addTransform(a, original, row, context, scope, { ...frame, targetId: id }, group, spec);
      }
      return;
    }
    // Parse structured row specifications before resolving text inside individual cells.
    const structured = ['addRow','updateRow','markRows','unmarkRows','addFilter','removeFilter','addSort','removeSort'].includes(a.type) || a.type === 'deleteRow' && (a.value ?? '').trim().startsWith('{');
    const template = a.value ?? '';
    if (a.type === 'openUrl') {
      if (template.length > 4096 || template !== template.trim() || !/^https?:\/\//i.test(template) || /[\u0000-\u0020\u007f\\]/.test(template) || template.includes('[[')) throw new Error('Use a literal HTTP or HTTPS URL');
      const url = new URL(template); if (!url.hostname || url.username || url.password || !['current','new'].includes(a.windowTarget ?? '')) throw new Error('Invalid URL navigation');
      if (!navigationHost) throw new Error('URL navigation host is not connected');
      navigationHost(template, a.windowTarget!); return;
    }
    if (a.type === 'openLinkInFrame') {
      if (!a.frameTarget || !['page','url','file'].includes(a.frameTarget.kind)) throw new Error('Invalid inline frame target');
      if (a.frameScope === 'parent') {
        if (!parentFrameHost) throw new Error('Parent frame navigation host is not connected');
        parentFrameHost(structuredClone(a.frameTarget)); return;
      }
      const target = a.targetId ? widgets.get(a.targetId) : undefined;
      if (!target?.inlineFrame || pageOwners.get(target.id) !== runtime.pageId) return;
      const scope = targetContexts(target.id, context)[0]; if (!scope || !validContext(scope)) return;
      const key = instanceKey(target.id, scope), states = runtime.frameNavigations ??= {}; runtime.scopes[key] = scope;
      states[key] = { target: structuredClone(a.frameTarget), revision: (states[key]?.revision ?? 0) + 1 }; return;
    }
    if (a.type === 'setVariable') { if (a.targetId && safeKey(a.targetId)) runtime.variables[a.targetId] = resolve(template, row, context, undefined, frame); return; }
    if (a.type === 'openPage') { const id = a.targetId || resolve(template, row, context, undefined, frame); if (doc.pages.some(p => p.id === id)) { changePage(id); runtime.pendingNavigation = id; } return; }
    const original = a.targetId ? widgets.get(a.targetId) : undefined;
    if (!original && a.type === 'rotate' && a.targetId && flatGroups.has(a.targetId)) {
      const members = flatGroups.get(a.targetId)!, first = members[0], owner = JSON.stringify(containers.get(first.id)), ancestors = JSON.stringify(owners.get(first.id));
      if (members.some(w => pageOwners.get(w.id) !== pageOwners.get(first.id) || JSON.stringify(containers.get(w.id)) !== owner || JSON.stringify(owners.get(w.id)) !== ancestors)) throw new Error('Rotate group members must be siblings in one coordinate frame');
      if (pageOwners.get(first.id) !== runtime.pageId) return;
      // A flat authoring group has no native Layer wrapper or nested layer ID.
      for (const scope of targetContexts(first.id, context, true)) addMotion(a, first, row, context, scope, frame, group, a.targetId);
      return;
    }
    if (!original) return;
    if (!motionScope && ['rotate', 'move', 'resize'].includes(a.type)) {
      for (const scope of targetContexts(original.id, context)) {
        if (a.type === 'rotate') addTransform(a, original, row, context, scope, frame, group);
        else addMotion(a, original, row, context, scope, frame, group);
      }
      return;
    }
    if (a.type === 'setFocus') {
      const scope = targetContexts(original.id, context)[0];
      if (scope && pageOwners.get(original.id) === runtime.pageId) setFocus(original, scope, Boolean(a.selectText), group);
      return;
    }
    if (a.type === 'setSelectedOption') {
      if (original.type !== 'select') return;
      // The native FCall evaluates every path target before applying the
      // individual setters in reverse order. Equality skips that target only.
      const targets = targetContexts(original.id, context).map(scope => ({ scope, value: resolve(template, row, context, scope, frame) }));
      const navigation = navigationVersion;
      for (let index = targets.length - 1; index >= 0; index--) {
        if (disposed || navigationVersion !== navigation) return;
        const { scope, value } = targets[index];
        if (validContext(scope)) setOptionValue(original, value, scope, group);
      }
      return;
    }
    for (const scope of motionScope ? [motionScope] : targetContexts(original.id, context)) {
    // Resolve against each actual target. Empty template target sets do not
    // evaluate a value or manufacture an override for a nonexistent instance.
    const value = structured ? template : resolve(template, row, context, scope, frame, a.type === 'setError' ? 'errorState' : undefined);
    const w = baseWidget(a.type === 'setText' && original.treeNode ? original.treeNode.label : original, scope);
    const beforeFit = ancestorBounds(original, scope);
    let visibilityChange: 'show'|'hide'|undefined, geometryChange: 'move'|'resize'|undefined;
    let fitNeeded = ['show', 'hide', 'toggle', 'move', 'resize'].includes(a.type);
    const apply = (value: Partial<Widget>) => patch(w, value, scope);
    switch (a.type) {
      case 'show': case 'hide': case 'toggle': {
        const hidden = a.type === 'toggle' ? !w.hidden : a.type === 'hide';
        if (Boolean(w.hidden) !== hidden) { apply({ hidden }); visibilityChange = hidden ? 'hide' : 'show'; }
        break;
      }
      case 'expandTreeNode': case 'collapseTreeNode': case 'toggleTreeNode': expandTree(original, a.type === 'toggleTreeNode' ? !w.treeNode?.expanded : a.type === 'expandTreeNode', scope); break;
      case 'enable': case 'disable': setEnabled(w, a.type === 'enable', scope); break;
      case 'setText': {
        if (w.layer || w.type === 'inlineFrame' || w.type === 'table' || w.type === 'menu' || w.type === 'tree') break;
        if (w.type === 'input' || w.type === 'textarea') {
          ensureTextCache(w, scope);
          const displayed = hint(w.id, scope), before = displayText(w, scope);
          // Native form Set Text returns from the whole target loop on equality.
          if (before === value || value === '' && displayed !== undefined) return;
          if (w.inputType === 'file' && value !== '') {
            const error = new Error('File inputs can only be cleared programmatically. Choose a local file using the file picker.');
            error.name = 'InvalidStateError'; throw error;
          }
          apply({ text: formValue(w, value) }); updateHint(w, scope, value === ''); repaintError(w, scope);
          if (w.inputType === 'file' && value === '') selectedFiles.delete(instanceKey(w.id, scope));
          if (textChanged(w.id, scope)) dispatchTextChange(w, scope, group);
        } else if (w.type !== 'select') {
          const native = supportsNativeText(w);
          const richText = native ? replaceNativePlain(w.richText, value) : w.richText ? replaceRichPlain(w.richText, value) : undefined;
          // The original sole-span, single-line path changes the existing leaf's
          // text without repainting its hyperlink. Structural replacements reset it.
          const key = instanceKey(w.id, scope), previousPaint = richText?.links?.length ? linkPaints.get(key) : undefined;
          if (richBeforeStyle.has(key)) {
            const baseline = richBeforeStyle.get(key);
            richBeforeStyle.set(key, native ? replaceNativePlain(baseline, value) : baseline ? replaceRichPlain(baseline, value) : undefined);
          }
          apply({ text: richText ? richPlain(richText) : value, ...(richText ? { richText } : {}) });
          if (previousPaint) linkPaints.set(key, previousPaint);
          if (native) { nativeTextSetters.add(key); autoFitText(original, scope, group); }
        }
        break;
      }
      case 'setError': setError(original, value === 'toggle' ? 'toggle' : Boolean(value) && value !== 'false', scope, group); break;
      case 'setSelected': {
        if (!['true', 'false', 'toggle'].includes(value)) throw new Error('Invalid selected state');
        setSelected(original, value === 'toggle' ? 'toggle' : value === 'true', scope, group);
        break;
      }
      case 'move': {
        const x = bound(a.x, w.x), y = bound(a.y, w.y); apply({ x, y });
        if (rotationLocations.has(instanceKey(w.id, scope))) rotationLocations.set(instanceKey(w.id, scope), { x, y });
        geometryChange = 'move'; break;
      }
      case 'resize': if (!['table','tableCell','menu','tree','treeNode'].includes(w.type)) {
        shapeCssSizes.delete(instanceKey(w.id, scope)); fittedPanelSizes.delete(instanceKey(w.id, scope));
        apply({ width: bound(a.x, w.width, 1), height: bound(a.y, w.height, 1), ...(w.type === 'panel' ? { fitToContent: false } : {}) }); geometryChange = 'resize';
      } else fitNeeded = false; break;
      case 'resizeItem': {
        const item = context[scope.length];
        if (!w.repeater || item?.repeaterId !== w.id || !scope.every((owner, index) => owner.rowId === context[index]?.rowId && owner.repeaterId === context[index]?.repeaterId)) break;
        const key = ensureRows(w, scope), size = runtime.itemLayouts[key][item.rowId];
        if (size) setItemSize(w, scope, item.rowId, bound(a.x, size.width, 0), bound(a.y, size.height, 0), group);
        break;
      }
      case 'setStyle': if (a.style) {
        const key = instanceKey(w.id, scope);
        if (!richBeforeStyle.has(key)) richBeforeStyle.set(key, w.richText ? structuredClone(w.richText) : undefined);
        runtime.richTextStyleOverrides ??= {};
        runtime.richTextStyleOverrides[key] = { ...runtime.richTextStyleOverrides[key], ...a.style };
        // Plain and rich text use the same atomic decoration conversion. Keep
        // original decoration in the cache so an invalid combined write can
        // restore it; do not corrupt that baseline with the raw booleans.
        const { underline: _underline, strikethrough: _strike, ...baselinePatch } = a.style;
        apply({ style: mergeStyle(w.style, baselinePatch), ...(w.richText ? { richText: richStyle(w.richText, baselinePatch) } : {}) });
      } break;
      case 'setState': case 'nextState': case 'previousState': {
        const states = w.states; if (!states?.length) break;
        const index = Math.max(0, states.findIndex(s => s.id === w.activeStateId));
        const next = a.type === 'setState' ? states.find(s => s.id === value || s.name === value) ?? (/^[1-9]\d*$/.test(value) ? states[Number(value) - 1] : undefined) : states[(index + (a.type === 'nextState' ? 1 : states.length - 1)) % states.length];
        if (next && next.id !== (states[index]?.id)) { apply({ activeStateId: next.id }); fitNeeded = true; } break;
      }
      case 'markRows': case 'unmarkRows': {
        if (!w.repeater) break;
        const key = ensureRows(w, scope), selected = selectedRows(JSON.parse(value), w, key, scope, row, context, frame, false);
        // Evaluate the complete rule before changing the set. Marking also works
        // during Item Loaded and deliberately does not request a row refresh.
        const marks = new Set(runtime.markedRowIds[key]);
        for (const id of selected) if (a.type === 'markRows') marks.add(id); else marks.delete(id);
        runtime.markedRowIds[key] = [...marks];
        break;
      }
      case 'setRepeaterPage': case 'setItemsPerPage': {
        if (!w.repeater) break;
        const key = ensureRows(w, scope); if (refreshing.has(key)) break;
        const paging = runtime.pagination[key];
        if (a.type === 'setItemsPerPage') {
          if (value === 'all') { delete paging.itemsPerPage; paging.page = 1; }
          else {
            if (paging.itemsPerPage === undefined) paging.page = 1;
            const size = Number(value); paging.itemsPerPage = Number.isNaN(size) ? 20 : Math.max(1, size);
          }
        } else if (paging.itemsPerPage !== undefined) {
          const last = Math.max(1, Math.ceil(runtime.committedItemCounts[key] / paging.itemsPerPage));
          if (value === 'next') { if (paging.page < last) paging.page++; }

          else if (value === 'first') paging.page = 1;
          else { const page = Number(value); paging.page = Number.isNaN(page) ? 1 : Math.max(1, Math.min(last, page)); }

        break;

        } else if (a.type === 'addSort') {
          if (keys.length !== 6 || keys.some(k => !['name','column','direction','dataType','toggle','removeOthers'].includes(k)) || !nameValid(spec.name)
            || typeof spec.column !== 'string' || !columns.includes(spec.column) || !['asc','desc'].includes(spec.direction)
            || !['number','text','textCase','natural','dateYMD','dateMDY','dateDMY'].includes(spec.dataType)

          const sorts = spec.removeOthers ? [] : runtime.querySorts[key].filter(sort => sort.column !== spec.column);
          if (sorts.length >= 100) throw new Error('Repeater sort limit of 100 reached');
          if (spec.removeOthers) delete runtime.sorts[key];
          runtime.querySorts[key] = [...sorts, { name: spec.name, column: spec.column, direction: spec.toggle && old ? old.direction === 'asc' ? 'desc' : 'asc' : spec.direction, dataType: spec.dataType }];
        } else {
          if (keys.length !== 1 || !(keys[0] === 'all' && spec.all === true || keys[0] === 'name' && nameValid(spec.name))) throw new Error('Invalid repeater rule removal');
          if (a.type === 'removeFilter') {
            if (spec.all) { runtime.queryFilters[key] = []; delete runtime.filters[key]; }

          } else {

            else if (spec.name !== '') runtime.querySorts[key] = runtime.querySorts[key].filter(sort => sort.name !== spec.name);
          }
        }
        runtime.pendingRepeaters[key] = { id: w.id, context: scope };
        break;
      }
      case 'addRow': case 'deleteRow': case 'sortRows': case 'filterRows': case 'updateRow': {
        if (!w.repeater) break;

          rows.push(Object.fromEntries(columns.map(c => [c, data[c] ?? ''])));
          ids.push(`row-${runtime.nextRowId++}`);
        } else if (a.type === 'updateRow') {

          if (!Object.keys(spec.values).length) break;

          // column can read an Item value changed on another selected source row.
          for (const [column, template] of Object.entries(spec.values)) for (const id of selected) {
            const index = ids.indexOf(id); if (index < 0) continue;
            const target = [...scope, { repeaterId: w.id, rowId: id }];
            Object.assign(rows[index], rowValues({ [column]: template }, columns, row, context, target, frame));
          }

            runtime.markedRowIds[key] = runtime.markedRowIds[key].filter(marked => marked !== id);
            forgetRow([...scope, { repeaterId: w.id, rowId: id }]);
          }
        } else if (a.type === 'sortRows') {
          if (!value.trim()) delete runtime.sorts[key];
          else { const i = value.lastIndexOf(':'), column = i < 0 ? value : value.slice(0, i), direction = i < 0 ? 'asc' : value.slice(i + 1); if (!columns.includes(column) || !['asc', 'desc'].includes(direction)) throw new Error('Invalid row sort'); runtime.sorts[key] = { column, direction }; }
        } else {
          if (!value.trim()) delete runtime.filters[key];
          else { const filter = JSON.parse(value); if (!filter || typeof filter !== 'object' || Array.isArray(filter) || Object.keys(filter).some(k => !['column', 'operator', 'value'].includes(k)) || !columns.includes(filter.column) || !['equals', 'notEquals', 'contains', 'greater', 'less', 'isEmpty'].includes(filter.operator) || typeof filter.value !== 'string') throw new Error('Invalid row filter'); runtime.filters[key] = filter; }
        }
        refreshRows(key);
        runtime.pendingRepeaters[key] = { id: w.id, context: scope };
        break;
      }

      const key = instanceKey(w.id, scope);
      if (geometryChange === 'resize') resizingRects.set(key, { x: w.x, y: w.y, width: w.width, height: w.height });

    }
    if (visibilityChange) visibilityEvent(w, visibilityChange, scope, group);
    }

    finally { geometryDepth--; }
  };
  const autoFitText = (w: Widget, context: RowContext, group: TaskGroup): void => {
    const initial = currentWidget(w, context);
    if (!supportsNativeText(initial) || !initial.shapeText || !initial.shapeText.autoFitWidth && !initial.shapeText.autoFitHeight || !controlHost?.measureShapeText) return;
    const navigation = navigationVersion, key = instanceKey(w.id, context);
    controlHost.flush();
    const measurement = controlHost.measureShapeText(w.id, context);
    if (!measurement || disposed || navigationVersion !== navigation || !validContext(context)) return;
    const current = currentWidget(w, context), old = measurement.rect;
    const width = current.shapeText!.autoFitWidth ? measurement.width : old.width;
    const height = current.shapeText!.autoFitHeight ? measurement.height : old.height;
    if (![width, height, old.x, old.y, old.width, old.height].every(Number.isFinite) || width < 0 || height < 0) throw new Error('Invalid shape text measurement');
    const deltaX = width - old.width, deltaY = height - old.height;
    if (deltaX || deltaY) {
      const beforeFit = ancestorBounds(w, context);
      // Original query.resize writes zero CSS but skips its logical cache when
      // BOTH requested axes are zero; retain that distinction for expressions.
      shapeCssSizes.set(key, { width, height });
      if (width || height) patch(w, { width, height }, context);
      resizingRects.set(key, { ...old });
      try {
        fitFrom(w, context, group, beforeFit, true);
        if (!disposed && navigationVersion === navigation && validContext(context)) geometryEvent(w, 'resize', context, group);
      } finally { resizingRects.delete(key); }
      if (disposed || navigationVersion !== navigation || !validContext(context)) return;
      const dx = measurement.textAlign === 'right' ? -deltaX : measurement.textAlign === 'center' ? -deltaX / 2 : 0;
      const dy = measurement.verticalAlign === 'bottom' ? -deltaY : measurement.verticalAlign === 'middle' ? -deltaY / 2 : 0;
      if (dx || dy) {
        const afterHandlers = currentWidget(w, context);
        patch(w, { x: afterHandlers.x + dx, y: afterHandlers.y + dy }, context);
        // Direct native MoveWidget has no second default fitParent callback.
        geometryEvent(w, 'move', context, group);
      }
    }
    if (!disposed && navigationVersion === navigation && validContext(context)) {
      const scope = structuredClone(widgetContext(w.id, context));
      autoFitted.set(key, { id: w.id, context: scope }); runtime.scopes[key] = scope;
    }
  };
  let textDepth = 0;
  const dispatchTextChange = (w: Widget, context: RowContext, group: TaskGroup): void => {
    if (++textDepth > 64 || ++group.textChanges > 10000) { textDepth--; throw new Error('Text change event cycle limit exceeded'); }
    try { dispatch(w.interactions, 'change', w.id, group, rowData(context), context); }
    finally { textDepth--; }
  };
  let optionDepth = 0;
  const dispatchOptionChange = (w: Widget, context: RowContext, group: TaskGroup, anchor?: { x: number; y: number }): void => {
    if (++optionDepth > 64 || ++group.optionChanges > 10000) { optionDepth--; throw new Error('Option change event cycle limit exceeded'); }
    try { dispatch(w.interactions, 'change', w.id, group, rowData(context), context, anchor); }
    finally { optionDepth--; }
  };
  const setOptionValue = (w: Widget, value: string, context: RowContext, group: TaskGroup): void => {
    const current = baseWidget(w, context), previous = optionValues(w, context);
    // Native jQuery returns null for an unselected single control, but [] for
    // multiple. Its abstract equality with a scalar therefore differs here.
    if (current.multiple ? previous.join(',') === value : previous.length > 0 && previous[0] === value) return;
    const matches = (current.options ?? []).flatMap((option, index) => option === value ? [index] : []);
    patch(w, { selectedIndices: current.multiple ? matches : matches.slice(0, 1) }, context);
    notify(); dispatchOptionChange(w, context, group, controlHost?.anchor?.(w.id, context, 'change'));
  };
  let errorDepth = 0;
  const setEnabled = (w: Widget, enabled: boolean, context: RowContext): void => {
    if (isUnplaced(w, context) || ['inlineFrame','table','tableCell','menu','tree','treeNode'].includes(w.type)) return;
    patch(w, { disabled: !enabled }, context);
    if (w.layer) for (const child of w.layer.widgets) setEnabled(child, enabled, context);
  };
  const setError = (w: Widget, value: boolean | 'toggle', context: RowContext, group: TaskGroup, anchor?: { x: number; y: number }): void => {
    if (disposed || !validContext(context) || ['inlineFrame','repeater','table','tableCell','menu','tree','treeNode'].includes(w.type)) return;
    const desired = value === 'toggle' ? !isError(w, context) : value;
    const navigation = navigationVersion, epoch = pageEpoch;
    if (!baseWidget(w, context).disabled) {
      if (++errorDepth > 64 || ++group.errorChanges > 10000) { errorDepth--; throw new Error('Error event cycle limit exceeded'); }
      try {
        const changed = isError(w, context) !== desired;
        patch(w, { error: desired }, context);
        if (changed) dispatch(w.interactions, desired ? 'errorSet' : 'errorRemoved', w.id, group, rowData(context), context, anchor);
        if (disposed || !validContext(context) || epoch !== pageEpoch || navigation !== navigationVersion) return;
        const key = instanceKey(w.id, context);
        if (hasInteractionStyle(w.id, 'error') || errorPaint.get(key)) errorPaint.set(key, desired);
        notify();
      } finally { errorDepth--; }
    }
    for (const child of [...(w.layer?.widgets ?? []), ...(w.states?.flatMap(state => state.widgets) ?? [])]) {
      if (disposed || !validContext(context) || navigation !== navigationVersion) return;
      if (w.layer && isUnplaced(child, context)) continue;
      setError(child, desired, context, group, anchor);
    }
  };
  let selectionDepth = 0;
  const selectionPeers = (w: Widget, context: RowContext, field: 'selectionGroup' | 'radioGroup'): Widget[] => {
    if (field === 'selectionGroup' && treeLabelOwner(w.id)) {
      const root = treeRoot(w.id);
      return [...widgets.values()].filter(other => treeLabelOwner(other.id) && treeRoot(other.id)?.id === root?.id && !isUnplaced(other, context));
    }
    const name = w[field]; if (!name) return [];
    const scope = widgetContext(w.id, context), ancestry = owners.get(w.id) ?? [];
    // This editor isolates template groups to their row. Native repeaters also
    // have a separate isolateSelection option, which RP import does not yet map.
    return [...widgets.values()].filter(other => other[field] === name && pageOwners.get(other.id) === pageOwners.get(w.id) && (field !== 'selectionGroup' || !isUnplaced(other, context))
      && (field !== 'radioGroup' || other.type === 'radio')
      && JSON.stringify(owners.get(other.id) ?? []) === JSON.stringify(ancestry)
      && JSON.stringify(widgetContext(other.id, context)) === JSON.stringify(scope));
  };
  const writeChecked = (w: Widget, value: boolean, context: RowContext) => {
    const current = baseWidget(w, context);
    patch(w, { checked: value, selected: selectedState(current) }, context);
  };
  const applySelected = (w: Widget, value: boolean, context: RowContext, group: TaskGroup, anchor?: { x: number; y: number }): void => {
    if (w.type === 'inlineFrame' || w.type === 'table' || w.type === 'menu' || w.type === 'tree') return;
    if (w.treeNode) { applySelected(w.treeNode.label, value, context, group, anchor); return; }
    if (disposed || !validContext(context) || baseWidget(w, context).disabled) return;
    if (w.layer) for (const child of w.layer.widgets) applySelected(child, value, context, group, anchor);
    const changed = selectedState(baseWidget(w, context)) !== value;
    if (++selectionDepth > 64 || ++group.selectionChanges > 10000) { selectionDepth--; throw new Error('Selection event cycle limit exceeded'); }
    try {
      if (value && !isCheck(w)) for (const other of selectionPeers(w, context, 'selectionGroup')) if (other.id !== w.id) applySelected(other, false, context, group, anchor);
      patch(w, { selected: value }, context); notify();
      if (changed) {
        const epoch = pageEpoch, navigation = navigationVersion;
        const source = w.interactions.length ? w : treeLabelOwner(w.id) ?? w;
        dispatch(source.interactions, 'selectedChange', source.id, group, rowData(context), context, anchor);
        if (!disposed && validContext(context) && epoch === pageEpoch && navigation === navigationVersion) dispatch(source.interactions, value ? 'select' : 'unselect', source.id, group, rowData(context), context, anchor);
      }
    } finally { selectionDepth--; }
  };
  const setSelected = (w: Widget, value: boolean | 'toggle', context: RowContext, group: TaskGroup, anchor?: { x: number; y: number }): void => {
    if (w.treeNode) { setSelected(w.treeNode.label, value, context, group, anchor); return; }
    const current = baseWidget(w, context), desired = value === 'toggle' ? !(isCheck(w) ? current.checked : current.selected) : value;
    if (isCheck(w)) {
      if (Boolean(current.checked) === desired) return;
      // The shipped query setter updates DOM checked before its style setter's
      // disabled guard. Browser radio grouping clears peer checked values only;
      // it does not dispatch those peers' selection events.
      if (w.type === 'radio' && desired) for (const peer of selectionPeers(w, context, 'radioGroup')) if (peer.id !== w.id) writeChecked(peer, false, context);
      writeChecked(w, desired, context); notify();
    }
    applySelected(w, desired, context, group, anchor);
  };
  const userCheck = (w: Widget, checked: boolean, context: RowContext, group: TaskGroup, anchor?: { x: number; y: number }): void => {
    if (w.type === 'radio') {
      const peers = selectionPeers(w, context, 'radioGroup');
      for (const peer of peers) if (peer.id !== w.id) writeChecked(peer, false, context);
      writeChecked(w, true, context);
      // Native click explicitly clears every named group member including itself,
      // so clicking an already selected named radio fires unselect then select.
      for (const peer of peers) applySelected(peer, false, context, group, anchor);
      applySelected(w, true, context, group, anchor);
    } else {
      const desiredStyle = !selectedState(baseWidget(w, context));
      writeChecked(w, checked, context); applySelected(w, desiredStyle, context, group, anchor);
    }
    notify();
  };
  const initializeLayerStates = (items: Widget[], group: TaskGroup, context: RowContext = []): void => {
    const records: { widget: Widget; selected: boolean; error: boolean; disabled: boolean }[] = [];
    const collect = (items: Widget[], withinLayer: boolean) => {
      for (const widget of items) {
        const inside = withinLayer || Boolean(widget.layer);
        if (inside && !isUnplaced(widget, context, true)) {
          const current = baseWidget(widget, context);
          records.push({ widget, selected: Boolean(current.selected), error: Boolean(current.error), disabled: Boolean(current.disabled) });
        }
        if (widget.layer) collect(widget.layer.widgets, true);
        for (const state of widget.states ?? []) collect(state.widgets, inside);
        // Repeater templates are initialized only in actual row instances.
      }
    };
    collect(items, false);
    // Native state dictionaries start empty. An authored Disabled declaration
    // is applied after Selected, not used as an earlier selection guard. A
    // parent's actual disable then blocks a later child's Selected declaration.
    for (const record of records) {
      const reset: Partial<Widget> = {};
      if (record.selected) reset.selected = false;
      if (record.error) reset.error = false;
      if (record.disabled) reset.disabled = false;
      if (Object.keys(reset).length) patch(record.widget, reset, context);
    }
    const navigation = navigationVersion;
    for (const record of records) {
      if (disposed || !validContext(context) || navigation !== navigationVersion) return;
      if (record.error) setError(record.widget, true, context, group);
      if (record.selected) setSelected(record.widget, true, context, group);
      if (record.disabled) setEnabled(record.widget, false, context);
    }
  };
  const loadWidgets = (items: Widget[], group: TaskGroup, context: RowContext = []): void => {
    const navigation = navigationVersion;
    for (const widget of items) {
      if (disposed || !validContext(context) || navigationVersion !== navigation) return;
      if (isUnplaced(widget, context, true)) continue;
      // Axure initializes hidden widgets and every panel state, not only the painted state.
      dispatch(widget.interactions, 'load', widget.id, group, rowData(context), context);
      if (navigationVersion !== navigation) return;
      if (widget.layer) loadWidgets(widget.layer.widgets, group, context);
      for (const state of widget.states ?? []) loadWidgets(state.widgets, group, context);
      if (widget.table) loadWidgets(tableCells(widget), group, context);
      if (widget.menu) loadWidgets(menuChildren(widget), group, context);
      if (widget.tree || widget.treeNode) loadWidgets(treeChildren(widget), group, context);
      if (widget.repeater) {
        const key = ensureRows(widget, context);
        runtime.pendingRepeaters[key] = { id: widget.id, context: widgetContext(widget.id, context) };
      }
    }
  };
  const loadEmptyTemplate = (items: Widget[], group: TaskGroup, context: RowContext, frame: EventFrame): void => {
    const navigation = navigationVersion;
    for (const widget of items) {
      if (disposed || !validContext(context) || navigationVersion !== navigation) return;
      if (isUnplaced(widget, context, true)) continue;
      dispatch(widget.interactions, 'load', widget.id, group, undefined, context, undefined, frame);
      if (navigationVersion !== navigation) return;
      if (widget.layer) loadEmptyTemplate(widget.layer.widgets, group, context, frame);
      for (const state of widget.states ?? []) loadEmptyTemplate(state.widgets, group, context, frame);
      if (widget.table) loadEmptyTemplate(tableCells(widget), group, context, frame);
      if (widget.menu) loadEmptyTemplate(menuChildren(widget), group, context, frame);
      if (widget.tree || widget.treeNode) loadEmptyTemplate(treeChildren(widget), group, context, frame);
      // No actual row exists. Do not initialize nested datasets or row DOM.
    }
  };
  const emptyTransitionCount = (key: string, active: number): number => {
    const paging = runtime.pagination[key], size = paging.itemsPerPage;
    if (size === undefined) return active;
    // This native lifecycle provider uses page * size, unlike visible bounds.
    // In particular, clearing the last page does not request an empty load.
    const start = Math.min(active, size * paging.page);
    return Math.min(active, start + size) - start;
  };
  const flushRepeaters = (group: TaskGroup): void => {
    if (!repeatersReady) return;
    const page = runtime.pageId, navigation = navigationVersion;
    // Mutual data-changing handlers may otherwise request one another forever.
    for (let passes = 0; !disposed && navigationVersion === navigation; passes++) {
      const entry = Object.entries(runtime.pendingRepeaters).find(([, pending]) => pageOwners.get(pending.id) === page && !isUnplaced(widgets.get(pending.id)!, pending.context, true));
      if (!entry) return;
      if (passes >= 1000) throw new Error('Repeater refresh loop stopped after 1000 refreshes');
      const [key, pending] = entry;
      delete runtime.pendingRepeaters[key];
      if (!validContext(pending.context)) continue;
      const widget = widgets.get(pending.id)!;
      const beforeFit = ancestorBounds(widget, pending.context);
      const previousItemCount = emptyTransitionCount(key, runtime.committedItemCounts[key]);
      // Intermediate mutations may temporarily shrink the data and restore it
      // in the same segment. Only the committed refresh may correct the page.
      refreshRows(key, true);
      // A native reset rebuilds each item from its template; data identity survives,
      // but temporary input, checked/disabled state and nested datasets do not.
      const persist = adaptiveRefreshes.delete(key);
      for (const rowId of runtime.sourceRowIds[key] ?? []) forgetRow([...pending.context, { repeaterId: widget.id, rowId }], persist);
      runtime.rowRevisions[key] = (runtime.rowRevisions[key] ?? 0) + 1;
      runtime.committedItemCounts[key] = runtime.filteredRowIds[key].length;
      runtime.committedVisibleCounts[key] = runtime.visibleRowIds[key].length;
      runtime.committedVisibleRowIds[key] = [...runtime.visibleRowIds[key]];
      resetItemLayout(widget, key);
      refreshing.add(key);
      refreshingGeometry.set(key, {});
      try {
        const rows = rowsFor(widget, pending.context);
        if (!rows.length && previousItemCount > 0 && emptyTransitionCount(key, runtime.filteredRowIds[key].length) === 0) {
          const frame: EventFrame = { pageEpoch, emptySource: true };
          dispatch(widget.interactions, 'itemLoad', widget.id, group, undefined, pending.context, undefined, frame);
          if (disposed || navigationVersion !== navigation) return;
          loadEmptyTemplate(widget.repeater!.template, group, pending.context, frame);
        }
        for (const item of rows) {
          if (disposed || navigationVersion !== navigation) return;
          if (!persist) initializeLayerStates(widget.repeater!.template, group, item.context);
          if (navigationVersion !== navigation) return;
          dispatch(widget.interactions, 'itemLoad', widget.id, group, item.data, item.context);
          if (navigationVersion !== navigation) return;
          loadWidgets(widget.repeater!.template, group, item.context);
        }
      } finally { refreshing.delete(key); refreshingGeometry.delete(key); }
      fitFrom(widget, pending.context, group, beforeFit);
      if (!disposed) notify();
    }
  };
  const loads = (group: TaskGroup): void => {
    for (let hops = 0; hops < 16 && !disposed; hops++) {
      const navigation = navigationVersion, page = doc.pages.find(p => p.id === runtime.pageId)!;
      // Native loadFinished and repeater.initRefresh are separate barriers.
      // Wait continuations release this synchronous initialization sequence.
      repeatersReady = false;
      if (!loadFinished) initializeLayerStates(page.widgets, group);
      if (!loadFinished && activeSelectionId) refreshLayerCaches(group);
      if (navigationVersion !== navigation) { commitNavigation(); continue; }
      run(page.interactions, 'load', group);
      if (navigationVersion === navigation) loadWidgets(page.widgets, group);
      if (navigationVersion === navigation && !loadFinished) {
        loadFinished = true;
        try { if (activeSelectionId) adaptiveChanged(group); }
        catch (error) {
          // A rejected initialization dispatch must not make a later retry
          // silently skip this barrier. Earlier committed action segments keep
          // their existing rollback contract; this only restores the phase.
          loadFinished = false;
          throw error;
        }
      }
      if (navigationVersion === navigation) { repeatersReady = true; flushRepeaters(group); }
      if (navigationVersion === navigation) { notify(); return; }
      commitNavigation();
    }
    if (!disposed) throw new Error('Page load navigation loop stopped after 16 transitions');
  };
  const settle = (group: TaskGroup): void => {
    if (disposed) return;
    if (navigationVersion === committedNavigation) flushRepeaters(group);
    if (navigationVersion !== committedNavigation) { commitNavigation(); loads(group); }
    const menuCount = menus.size;
    for (const [id, pending] of menus) if (pending.frame.pageEpoch !== pageEpoch || !validContext(pending.context)) menus.delete(id);
    if (menus.size !== menuCount) { publishMenus(); notify(); }
  };
  const visibleControl = (w: Widget, context: RowContext): boolean => {
    if (pageOwners.get(w.id) !== runtime.pageId || baseWidget(w, context).hidden || isUnplaced(w, context, true)) return false;
    let childId = w.id, parent = containers.get(w.id);
    while (parent) {
      const ancestor = baseWidget(widgets.get(parent.id)!, context);
      if (ancestor.hidden || parent.stateId && parent.stateId !== (ancestor.activeStateId ?? ancestor.states?.[0]?.id)) return false;
      if (ancestor.treeNode && ancestor.treeNode.label.id !== childId && !ancestor.treeNode.expanded) return false;
      childId = ancestor.id;
      parent = containers.get(parent.id);
    }
    return context.every((row, index) => runtime.committedVisibleRowIds[instanceKey(row.repeaterId, context.slice(0, index))]?.includes(row.rowId));
  };
  const disabledControl = (w: Widget, context: RowContext): boolean => {
    if (baseWidget(w, context).disabled) return true;
    let parent = containers.get(w.id);
    while (parent) { const ancestor = baseWidget(widgets.get(parent.id)!, context); if (!['layer','table','menu','tree','treeNode'].includes(ancestor.type) && ancestor.disabled) return true; parent = containers.get(parent.id); }
    return false;
  };
  let focusDepth = 0;
  const focusChanged = (w: Widget, focused: boolean, context: RowContext, group: TaskGroup, anchor?: { x: number; y: number }, suppressEvents = false, valid = true): void => {
    if (disposed || !validContext(context) || disabledControl(w, context)) return;
    if (++focusDepth > 64 || ++group.focusChanges > 10000) { focusDepth--; throw new Error('Focus event cycle limit exceeded'); }
    try {
      const navigation = navigationVersion;
      ensureTextCache(w, context); setInteractionState(w.id, { focused }, context);
      if (!suppressEvents && visibleControl(w, context)) dispatch(w.interactions, focused ? 'focus' : 'blur', w.id, group, rowData(context), context, anchor);
      if (navigation !== navigationVersion || !validContext(context) || disposed) return;
      const state = ensureHint(w, context);
      if (state) {
        if (focused && w.hintHideOn === 'focus' && state.active) updateHint(w, context, false);
        else if (!focused && displayText(w, context) === '' && valid) updateHint(w, context, true);
        repaintError(w, context); notify();
      }
    } finally { focusDepth--; }
  };
  const setFocus = (w: Widget, context: RowContext, selectText: boolean, group: TaskGroup): void => {
    if (w.layer) { const first = w.layer.widgets[0]; if (first) setFocus(first, context, selectText, group); return; }
    if (disposed || !validContext(context) || isUnplaced(w, context, true)) return;
    const previous = focusBridge, bridge = { group, target: instanceKey(w.id, context), nativeFocus: 0 } as NonNullable<typeof focusBridge>;
    focusBridge = bridge;
    const navigation = navigationVersion;
    try {
      controlHost?.flush();
      // The original query runs triggerHandler('focus') before jQuery.focus().
      // Even an already focused element receives both rounds of handlers.
      focusChanged(w, true, context, group, controlHost?.anchor?.(w.id, context));
      if (disposed || navigation !== navigationVersion || !validContext(context)) return;
      controlHost?.flush();
      if (controlHost) controlHost.focus(w.id, context);
      else if (!disabledControl(w, context) && visibleControl(w, context)) {
        const prior = focusedControl;
        if (prior && instanceKey(prior.id, prior.context) !== bridge.target) {
          const old = widgets.get(prior.id); if (old) focusChanged(old, false, prior.context, group);
        }
        focusedControl = { id: w.id, context };
      }
      if (bridge.failure !== undefined) throw bridge.failure;
      if (disposed || navigation !== navigationVersion || !validContext(context)) return;
      if (!bridge.nativeFocus) focusChanged(w, true, context, group, controlHost?.anchor?.(w.id, context));
      controlHost?.flush();
      if (selectText && hint(w.id, context) === undefined && (w.type === 'input' || w.type === 'textarea')) controlHost?.selectText(w.id, context);
      if (bridge.failure !== undefined) throw bridge.failure;
    } finally { focusBridge = previous; }
  };
  const enqueue = (operation: (group: TaskGroup) => void): Promise<void> => {
    let resolve!: () => void, reject!: (error: unknown) => void;
    const done = new Promise<void>((success, failure) => { resolve = success; reject = failure; });
    const group: TaskGroup = { pending: 0, settled: false, selectionChanges: 0, optionChanges: 0, errorChanges: 0, textChanges: 0, focusChanges: 0, itemResizes: 0, visibilityChanges: 0, geometryChanges: 0, adaptiveChanges: 0, resolve, reject };
    queueTask(group, () => operation(group));
    return done;
  };
  const cancelDrag = () => {
    const drag = runtime.motionState?.drag; if (!drag) return;
    drag.started = false; drag.hasDragged = false; drag.movedWidgets = {};
  };
  const beginDrag = (id: string, x: number, y: number, context: RowContext = [], time = Date.now()): boolean => {
    if (![x, y, time].every(Number.isFinite)) throw new Error('Drag coordinates and time must be finite');
    const original = widgets.get(id);
    if (disposed || runtime.motionState?.drag?.started || !original || pageOwners.get(id) !== runtime.pageId || !validContext(context) || !visibleControl(original, context) || disabledControl(original, context)) return false;
    const w = currentWidget(original, context);
    if (w.hidden || w.disabled || !w.interactions.some(item => item.enabled && ['dragStart','drag','dragDrop','swipeLeft','swipeRight','swipeUp','swipeDown'].includes(item.event))) return false;
    (runtime.motionState ??= {}).drag = { sourceId: id, context: structuredClone([...context]), started: true, hasDragged: false, startX: x, startY: y, currentX: x, currentY: y, startTime: time, movedWidgets: {} };
    return true;
  };
  const dragMove = (x: number, y: number, time = Date.now()): Promise<void> => {
    if (![x, y, time].every(Number.isFinite)) return Promise.reject(new Error('Drag coordinates and time must be finite'));
    return enqueue(group => {
      const drag = runtime.motionState?.drag, original = drag?.sourceId && widgets.get(drag.sourceId), context = drag?.context ?? [];
      if (!drag?.started || !original || !validContext(context) || pageOwners.get(original.id) !== runtime.pageId) { cancelDrag(); return; }
      drag.deltaX = x - drag.currentX!; drag.deltaY = y - drag.currentY!; drag.currentX = x; drag.currentY = y; drag.currentTime = time;
      const first = !drag.hasDragged; drag.hasDragged = true;
      if (first) dispatch(original.interactions, 'dragStart', original.id, group, rowData(context), context);
      if (runtime.motionState?.drag === drag && drag.started) dispatch(original.interactions, 'drag', original.id, group, rowData(context), context);
      settle(group); notify();
    });
  };
  const endDrag = (time = Date.now()): Promise<void> => {
    if (!Number.isFinite(time)) return Promise.reject(new Error('Drag time must be finite'));
    return enqueue(group => {
      const drag = runtime.motionState?.drag, original = drag?.sourceId && widgets.get(drag.sourceId), context = drag?.context ?? [];
      if (!drag?.started) return;
      if (drag.hasDragged && original && validContext(context) && pageOwners.get(original.id) === runtime.pageId) {
        drag.currentTime = time;
        dispatch(original.interactions, 'dragDrop', original.id, group, rowData(context), context);
        if (time - drag.startTime! < 1000) {
          const dx = drag.currentX! - drag.startX!, dy = drag.currentY! - drag.startY!;

    },
    bindControls: host => { controlHost = host; },
    execute: (interaction, row, context = []) => enqueue(group => { run([interaction], interaction.event, group, row, context); settle(group); }),

    }),

    }),

    select: (id, value, context = [], anchor) => enqueue(group => { const w = widgets.get(id); if (!w || !validContext(context)) return; setSelected(w, value, context, group, anchor); settle(group); }),
    check: (id, checked, context = [], anchor) => enqueue(group => { const w = widgets.get(id); if (!w || !isCheck(w) || !validContext(context) || baseWidget(w, context).disabled) return; userCheck(w, checked, context, group, anchor); settle(group); }),

    start: () => enqueue(loads),
    navigate: id => enqueue(group => { if (!doc.pages.some(p => p.id === id)) return; changePage(id); commitNavigation(); notify(); loads(group); }),

    }),
    dispose: () => { disposed = true; sourceMoves.clear(); if (runtime.motionState) { delete runtime.motionState.drag; if (!Object.keys(runtime.motionState).length) delete runtime.motionState; } for (const job of motions.values()) job.status = 'done'; commitMotions(); motionGenerations.clear(); rotationTargets.clear(); rotationLocations.clear(); resizeTargets.clear(); transformZeroSizes.clear(); transformBatches.clear(); transformRequests.clear(); transformMoveResults.clear(); groupAngles.clear(); clearShapeTextState(); notify = () => {}; controlHost = undefined; navigationHost = undefined; parentFrameHost = undefined; focusedControl = undefined; menuLayers.clear(); interactionStates.clear(); linkPaints.clear(); adaptiveTextStyles.clear(); richBeforeStyle.clear(); adaptiveRefreshes.clear(); repeaterInstances.clear(); persistedHintPaint.clear(); hintStates.clear(); errorPaint.clear(); textChangeValues.clear(); selectedFiles.clear(); initialControlValues.clear(); valueControls.clear(); cancelWaits(); deferredWaitCancellations.clear(); cancelCaseMenus(); },

}

/** DOM-only renderer. No HTML interpretation, dynamic code evaluation, or remote resource loading. */
export function mountOfflinePrototype(doc: PrototypeDocument, runtime: OfflineRuntime, host: HTMLElement, pathData?: (path: VectorPath, width: number, height: number) => string, options: { frames?: InlineFrameHost; embedded?: boolean; previewImages?: boolean; shadows?: ShadowRenderer; gradients?: GradientRenderer; lines?: LineStyleRenderer; shapeText?: ShapeTextRenderer; shapeTextMeasure?: import('./shape-text-measure').ShapeTextMeasurer; dragInput?: typeof import('./drag-input').bindRuntimeDrag } = {}): void {
  // Render identity only: getRandomValues also works on ordinary HTTP pages,
  // unlike randomUUID, which browsers restrict to secure contexts.
  const shadowScope = Array.from(crypto.getRandomValues(new Uint32Array(4)), value => value.toString(16)).join('-');
  if (!options.embedded) runtime.bindNavigation((url, target) => { if (target === 'current') window.location.assign(url); else window.open(url, '_blank', 'noopener,noreferrer'); });
  const decoration = (style: Pick<WidgetStyle, 'underline' | 'strikethrough'>) => style.underline ? style.strikethrough ? 'underline line-through' : 'underline' : style.strikethrough ? 'line-through' : 'none';
  const releasePointer = () => runtime.releasePointerStyles();
  window.addEventListener('mouseup', releasePointer); window.addEventListener('pointercancel', releasePointer); window.addEventListener('blur', releasePointer); window.addEventListener('dragend', releasePointer);
  const mergeStyle = (base: Widget['style'], value?: Partial<Widget['style']>): Widget['style'] => {
    const merged = { ...base, ...value };
    if (value && Object.hasOwn(value, 'fill') && !Object.hasOwn(value, 'fillGradient')) delete merged.fillGradient;

      indexShapes(children, ['panel', 'repeater'].includes(widget.type) ? [...parents, widget.id] : parents);
    }
  };
  for (const page of doc.pages) indexShapes(page.widgets);
  const measureShapeText = (id: string, context: RowContext) => {
    const source = shapeSources.get(id), helper = options.shapeTextMeasure;
    if (!source || !helper) return;
    const effective = (original: Widget, parentDisabled = false) => runtime.widget(original, context, parentDisabled);
    const parentDisabled = source.parents.some(parent => effective(shapeSources.get(parent)!.widget).disabled);
    const widget = effective(source.widget, parentDisabled);
    if (widget.richText) widget.richText = runtime.richText(widget, undefined, context);
    else if (!runtime.hasOverride(id, 'text', context)) widget.text = runtime.text(widget.text, undefined, context);
    const key = JSON.stringify(context);
    const owner = [...nodes.values()].find(node => node.isConnected && node.dataset.widgetId === id && node.dataset.rowContext === key);
    const size = helper.measure(widget, owner, document, { plainSetter: runtime.nativeTextSet(id, context) });
    return size && { ...size, rect: { x: widget.x, y: widget.y, width: widget.width, height: widget.height }, textAlign: widget.style.textAlign, verticalAlign: widget.style.verticalAlign ?? 'top' };
  };
  runtime.bindControls({ flush: safeRender, measureShapeText, anchor: (id, context, event = 'focus') => { const element = findControl(id, context); if (!element) return; const rect = element.getBoundingClientRect(), point = { x: rect.left, y: rect.bottom }; sourceFocus.set(sourceFocusKey(id, event, point), element); if (sourceFocus.size > 256) sourceFocus.delete(sourceFocus.keys().next().value!); return point; }, focus: (id, context) => findControl(id, context)?.focus(), selectText: (id, context) => { const element = findControl(id, context); if (element instanceof HTMLInputElement || element instanceof HTMLTextAreaElement) element.select(); } });
  const schedule = () => { if (drag?.hasActiveDrag() && !runtime.state().motionState?.drag?.started) drag.cancel(); if (!renderPending) { renderPending = true; queueMicrotask(safeRender); } };
  runtime.subscribe(schedule); report(runtime.start()); safeRender();
  window.addEventListener('resize', () => report(runtime.setViewport({ width: window.innerWidth, height: window.innerHeight, clampToPage: true }).then(safeRender)));
}
